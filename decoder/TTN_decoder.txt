function readU16BE(bytes, i) {
  return (bytes[i] << 8) | bytes[i + 1];
}

function readI16BE(bytes, i) {
  let v = readU16BE(bytes, i);
  if (v & 0x8000) v -= 0x10000;
  return v;
}

function readU32BE(bytes, i) {
  return (
    (bytes[i] * 16777216) +
    (bytes[i + 1] << 16) +
    (bytes[i + 2] << 8) +
    bytes[i + 3]
  ) >>> 0;
}

function decodeUplink(input) {
  const b = input.bytes;

  if (input.fPort !== 1) {
    return { errors: ["Unsupported FPort"] };
  }

  if (b.length !== 16) {
    return { errors: ["Invalid payload length"] };
  }

  const version = b[0];
  if (version !== 3) {
    return { errors: ["Unsupported payload version"] };
  }

  const flags = b[1];

  const sht30_present    = (flags & 0x01) !== 0;
  const moisture_present = (flags & 0x02) !== 0;
  const water_line_high  = (flags & 0x04) !== 0;
  const battery_valid    = (flags & 0x08) !== 0;
  const battery_low      = (flags & 0x10) !== 0;
  const battery_critical = (flags & 0x20) !== 0;
  const solar_present    = (flags & 0x40) !== 0;
  const charging         = (flags & 0x80) !== 0;

  const uptime_min   = readU16BE(b, 2);
  const water_pulses = readU32BE(b, 4);
  const tempRaw      = readI16BE(b, 8);
  const humRaw       = readU16BE(b, 10);
  const moistRaw     = readU16BE(b, 12);
  const battRaw      = readU16BE(b, 14);

  return {
    data: {
      payload_version: version,
      sht30_present,
      moisture_present,
      water_line_high,
      battery_valid,
      battery_low,
      battery_critical,
      solar_present,
      charging,
      uptime_min,
      water_pulses,
      water_liters: water_pulses / 87.0,
      temperature_c: tempRaw === -32768 ? null : tempRaw / 100.0,
      humidity_rh: humRaw === 0xFFFF ? null : humRaw / 100.0,
      moisture_raw: moistRaw === 0xFFFF ? null : moistRaw,
      battery_mv: battRaw === 0xFFFF ? null : battRaw
    }
  };
}
